set(PROJECT_NAME MemoryProject)

if (TARGET ${PROJECT_NAME})
    message(STATUS "# ${PROJECT_NAME} target is already defined. Skipping build configuration.")
    
    return()
endif()

cmake_minimum_required(VERSION 3.20)
project(${PROJECT_NAME} LANGUAGES CXX)

# ------ Create the library ------ #
include(${CMAKE_SOURCE_DIR}/external/CMake/Library.cmake)

set(LIBRARY_TYPE "SHARED" CACHE STRING "Build a shared library instead of a static one")

set(MAIN_DIR ${CMAKE_CURRENT_SOURCE_DIR})
set(HEADER_DIR ${MAIN_DIR}/include)
set(SOURCE_DIR ${MAIN_DIR}/src)
set(EXTERNAL_DIR ${CMAKE_SOURCE_DIR}/external)

set(INCLUDE_DIR_LIST
    ${MAIN_DIR}/include
)

set(HEADER_LIST
    ${HEADER_DIR}/Memory.h
    ${HEADER_DIR}/Core.h
    ${HEADER_DIR}/TimeLimit.h

    ${HEADER_DIR}/Accessor/IAccessor.h
    ${HEADER_DIR}/Accessor/Accessor.h

    ${HEADER_DIR}/Collector/GarbageCollector.h

    ${HEADER_DIR}/Storage/IStorage.h
    ${HEADER_DIR}/Storage/Array.h
    ${HEADER_DIR}/Storage/ArrayBudget.h
    ${HEADER_DIR}/Storage/Pool.h
    ${HEADER_DIR}/Storage/PoolBudget.h
    ${HEADER_DIR}/Storage/StorageManager.h

    ${HEADER_DIR}/Pointer/BasePtr.h
    ${HEADER_DIR}/Pointer/ObjectPtr.h
    ${HEADER_DIR}/Pointer/RefPtr.h
    ${HEADER_DIR}/Pointer/RootPtr.h
)

set(SOURCE_LIST
    ${SOURCE_DIR}/Core.cpp
    ${SOURCE_DIR}/TimeLimit.cpp

    ${SOURCE_DIR}/Collector/GarbageCollector.cpp

    ${SOURCE_DIR}/Storage/ArrayBudget.cpp
    ${SOURCE_DIR}/Storage/PoolBudget.cpp
    ${SOURCE_DIR}/Storage/StorageManager.cpp
)

source_group(TREE "${HEADER_DIR}" PREFIX "include" FILES ${HEADER_LIST})
source_group(TREE "${SOURCE_DIR}" PREFIX "src" FILES ${SOURCE_LIST})

create_library(
    LIBRARY_TYPE ${LIBRARY_TYPE}
    TARGET_NAME ${PROJECT_NAME}
    PUBLIC_HEADERS ${HEADER_LIST}
    PRIVATE_SOURCES ${SOURCE_LIST}
)

install_library(
    TARGET_NAME ${PROJECT_NAME}
    INCLUDE_DIRS ${INCLUDE_DIR_LIST}
)

# ------ Set the project options ------ #
include(${CMAKE_SOURCE_DIR}/external/CMake/Project.cmake)

project_options()

target_include_directories(${PROJECT_NAME} PUBLIC ${HEADER_DIR} ${EXTERNAL_DIR})
target_link_libraries(${PROJECT_NAME} PUBLIC ReflectionProject LogProject)

# ------ Add the sub modules ------ #
message(STATUS "- Add the Sub Project")

set(REFLECTION_DEMO_FILE OFF)
add_subdirectory(${CMAKE_SOURCE_DIR}/external/Reflection Reflection)

set(LOG_DEMO_FILE OFF)
add_subdirectory(${CMAKE_SOURCE_DIR}/external/Log Log)

set(CONTAINER_DEMO_FILE OFF)
add_subdirectory(${CMAKE_SOURCE_DIR}/external/Container Container)

# ------ Create the Demofile ------ #
option(MEMORY_DEMO_FILE "Build the demo file" ON)

if (MEMORY_DEMO_FILE)
    message(STATUS "# Create the demofile project")

    add_subdirectory(${MAIN_DIR}/demofile)
endif()
